const express = require("express");
const app = express();

app.use(express.json());

let state = {
  text: "натурал",
  until: null,
  updatedAt: Date.now()
};

// получить текущее состояние
app.get("/state", (req, res) => {
  res.json(state);
});

// обновить состояние (админка)
app.post("/update", (req, res) => {
  state = {
    ...req.body,
    updatedAt: Date.now()
  };
  res.json({ ok: true });
});

// авто-сброс по времени
setInterval(() => {
  if (state.until && Date.now() > state.until) {
    state.text = "натурал";
    state.until = null;
  }
}, 2000);

app.listen(3000, () => {
  console.log("Server running");
});
